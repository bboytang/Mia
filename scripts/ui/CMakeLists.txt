cmake_minimum_required(VERSION 3.16)
project(youmi_ui_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT LVGL_SOURCE)
    message(FATAL_ERROR "Pass -DLVGL_SOURCE=/path/to/lvgl-v9.5.0 (85aa60d)")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE}" lvgl-build)
set(BOARD_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../main/boards/folotoy/ai-passport")
add_executable(youmi-ui-preview preview.cc "${BOARD_DIR}/youmi_ui_font.c")
target_include_directories(youmi-ui-preview PRIVATE "${BOARD_DIR}")
target_link_libraries(youmi-ui-preview PRIVATE lvgl::lvgl)
target_compile_options(youmi-ui-preview PRIVATE -Wall -Wextra -Werror)
